```typescript
// src/utils/DependencyTracker.ts
export class DependencyTracker {
    private static dependencies: Map<string, string[]> = new Map();
    
    static addDependency(from: string, to: string): void {
        if (!this.dependencies.has(from)) {
            this.dependencies.set(from, []);
        }
        this.dependencies.get(from)?.push(to);
    }
    
    static logDependencies(): void {
        console.log("当前依赖关系图:");
        this.dependencies.forEach((deps, module) => {
            console.log(`${module} 依赖于: ${deps.join(', ')}`);
        });
    }
}
```

### 第三阶段：简单事件机制 🔄

1. **创建最小化事件总线**

```typescript
// src/core/EventBus.ts
export class EventBus {
    private static instance: EventBus;
    private listeners: Map<string, Function[]> = new Map();
    
    static getInstance(): EventBus {
        if (!EventBus.instance) {
            EventBus.instance = new EventBus();
        }
        return EventBus.instance;
    }
    
    subscribe(event: string, callback: Function): void {
        if (!this.listeners.has(event)) {
            this.listeners.set(event, []);
        }
        this.listeners.get(event)?.push(callback);
    }
    
    publish(event: string, ...args: any[]): void {
        const callbacks = this.listeners.get(event) || [];
        callbacks.forEach(callback => callback(...args));
    }
}
```

```typescript
// src/core/ServiceLocator.ts
export class ServiceLocator {
    private static instance: ServiceLocator;
    private services: Map<string, any> = new Map();
    
    static getInstance(): ServiceLocator {
        if (!ServiceLocator.instance) {
            ServiceLocator.instance = new ServiceLocator();
        }
        return ServiceLocator.instance;
    }
    
    register<T>(serviceId: string, instance: T): void {
        this.services.set(serviceId, instance);
    }
    
    get<T>(serviceId: string): T | null {
        return this.services.get(serviceId) || null;
    }
}
```

### 第六阶段：轻量级依赖注册和解析 💉

1. **扩展服务定位器支持工厂函数**

```typescript
// 扩展 ServiceLocator
export class ServiceLocator {
    // 现有代码...
    private factories: Map<string, () => any> = new Map();
    
    registerFactory<T>(serviceId: string, factory: () => T): void {
        this.factories.set(serviceId, factory);
    }
    
    get<T>(serviceId: string): T | null {
        // 先检查已存在的实例
        if (this.services.has(serviceId)) {
            return this.services.get(serviceId);
        }
        
        // 检查工厂函数
        const factory = this.factories.get(serviceId);
        if (factory) {
            const instance = factory();
            this.services.set(serviceId, instance);
            return instance;
        }
        
        return null;
    }
}
```

2. **使用工厂解决循环依赖**

## 4. 变更的好处与优势 🌟

以这种循序渐进的方式改造代码有以下优势：

1. **小步变更**：每个改动都很小，易于理解和实施
2. **持续可用**：每一步完成后系统仍然可以正常工作
3. **易于回滚**：如果某一步出现问题，容易回退到上一个工作状态
4. **学习曲线平缓**：初学者可以逐步掌握设计模式和技术
5. **增量测试**：每一步都可以单独测试，降低风险
