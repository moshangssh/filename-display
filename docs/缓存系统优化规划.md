# TitleExtractor 缓存系统优化规划

## 当前缓存系统问题

当前的缓存实现存在以下主要问题：

1. **架构过度复杂**
   - 多个缓存相关服务：`FileDisplayCache`、`CacheManager`、`ExtensionCacheService`等
   - 缓存策略、存储、清理分散在多个文件和类中
   - 职责边界不清晰，导致维护困难

2. **性能瓶颈**
   - 缓存键设计不合理，导致查找效率低
   - 序列化与反序列化开销大
   - 缓存更新频繁，导致性能损耗

3. **内存占用高**
   - 缓存条目过多且无有效清理策略
   - 重复数据存储在不同缓存中
   - 缓存管理结构本身开销大

4. **可靠性问题**
   - 缓存同步机制不完善，可能导致状态不一致
   - 缓存加载和持久化过程缺少错误恢复机制
   - 边缘情况处理不充分

## 优化目标

1. **简化架构**：将多个缓存相关服务整合为单一、统一的缓存服务
2. **提高性能**：优化缓存查询和更新性能，减少不必要的操作
3. **降低内存占用**：优化缓存存储结构，实现更高效的缓存清理
4. **增强可靠性**：改进缓存同步和持久化机制，确保数据一致性
5. **提高可测试性**：设计更易于测试的缓存系统

class CacheSerializer {
  // 增量序列化
  static serializeIncremental(cache: Map<string, any>, lastPersistTime: number): string {
    // 只序列化上次持久化后变更的数据
    const changedData = new Map();
    for (const [key, entry] of cache.entries()) {
      if (entry.lastModified > lastPersistTime) {
        changedData.set(key, entry);
      }
    }
    return JSON.stringify(Array.from(changedData.entries()));
  }
  
  // 增量反序列化
  static deserializeIncremental(serialized: string, existingCache: Map<string, any>): Map<string, any> {
    try {
      const parsed = JSON.parse(serialized);
      for (const [key, value] of parsed) {
        existingCache.set(key, value);
      }
      return existingCache;
    } catch (e) {
      console.error("缓存反序列化失败:", e);
      return existingCache; // 返回现有缓存，避免完全失败
    }
  }
}

// 缓存变更事件发布器
class CacheChangePublisher {
  private listeners: Map<string, Function[]> = new Map();
  
  // 添加变更监听器
  subscribe(domain: string, callback: (key: string, value: any) => void): void {
    if (!this.listeners.has(domain)) {
      this.listeners.set(domain, []);
    }
    this.listeners.get(domain)?.push(callback);
  }
  
  // 发布变更事件
  publish(domain: string, key: string, value: any): void {
    this.listeners.get(domain)?.forEach(callback => {

class IncrementalPersistenceManager {
  private lastPersistTime: number = 0;
  private isDirty: boolean = false;
  private persistInterval: number = 5 * 60 * 1000; // 5分钟
  private plugin: Plugin;
  
  constructor(plugin: Plugin) {
    this.plugin = plugin;
    
    // 设置自动持久化
    this.setupAutoPersist();
  }
  
  // 设置自动持久化
  private setupAutoPersist(): void {
    setInterval(() => {
      if (this.isDirty) {
        this.persistCache();
      }
    }, this.persistInterval);
  }
  
  // 标记为已修改
  markDirty(): void {
    this.isDirty = true;
  }
  
  // 持久化缓存
  async persistCache(): Promise<boolean> {
    if (!this.isDirty) {
      return true; // 没有变更，不需要持久化
    }
    
    try {
      // 获取需要持久化的域缓存
      const persistableData: Record<string, any> = {};
      
      // 只持久化上次持久化后变更的数据
      // ...实现增量持久化逻辑
      
      // 写入数据
      await this.plugin.saveData(persistableData);
      
      this.lastPersistTime = Date.now();
      this.isDirty = false;
      return true;
    } catch (e) {
      console.error("缓存持久化失败:", e);
      return false;
    }
  }
  
  // 加载缓存
  async loadCache(): Promise<Record<string, any>> {

### 5. 缓存服务实现

将上述所有概念整合到最终的缓存服务实现中：

```typescript
export class CacheService implements ICacheService {
  // 域缓存映射
  private caches: Map<string, DomainCache> = new Map();
  
  // 缓存管理组件
  private strategyManager: CacheStrategyManager;
  private sizeManager: AdaptiveSizeManager;
  private changePublisher: CacheChangePublisher;
  private persistenceManager: IncrementalPersistenceManager;
  private statsCollector: StatsCollector;
  
  constructor(private plugin: Plugin, private settings: TitleExtractorSettings) {
    // 初始化各管理组件
    this.strategyManager = new CacheStrategyManager(settings);
    this.sizeManager = new AdaptiveSizeManager();
    this.changePublisher = new CacheChangePublisher();
    this.persistenceManager = new IncrementalPersistenceManager(plugin);
    this.statsCollector = new StatsCollector();
    
    // 初始化缓存域
    this.initializeDomains();
    
    // 加载缓存
    this.load();
  }
  
  // 获取缓存项
  get<T>(domain: string, key: string): T | null {
    const cache = this.getOrCreateDomainCache(domain);
    const entry = cache.get(key);
    
    if (entry) {
      // 记录命中统计
      this.statsCollector.recordHit(domain);
      
      // 更新访问元数据
      const strategy = this.strategyManager.getStrategy(domain);
      strategy.updateMetadata(entry);
      
      return entry.value as T;
    }
    
    // 记录未命中统计
    this.statsCollector.recordMiss(domain);
    return null;
  }
  
  // 设置缓存项
  set<T>(domain: string, key: string, value: T, options?: CacheOptions): void {
    const cache = this.getOrCreateDomainCache(domain);
    
    // 检查是否需要清理
    const maxSize = this.getMaxSizeForDomain(domain);
    if (cache.size >= maxSize) {
      const strategy = this.strategyManager.getStrategy(domain);
      const keysToEvict = strategy.shouldEvict(cache, key);
      
      // 清理选定的键
      for (const keyToEvict of keysToEvict) {
        cache.delete(keyToEvict);
      }
    }
    
    // 创建并存储缓存条目
    const entry = CacheEntryFactory.create(value, options);
    cache.set(key, entry);
    
    // 标记缓存已修改
    this.persistenceManager.markDirty();
    
    // 发布变更事件
    this.changePublisher.publish(domain, key, value);
  }
  
  // 检查键是否存在
  has(domain: string, key: string): boolean {
    const cache = this.caches.get(domain);
    return cache ? cache.has(key) : false;
  }
  
  // 使缓存项无效
  invalidate(domain: string, key?: string): void {
    if (domain && key) {
      // 使特定键无效
      const cache = this.caches.get(domain);
      if (cache) {
        cache.delete(key);
        this.persistenceManager.markDirty();
      }
    } else if (domain) {
      // 使整个域无效
      const cache = this.caches.get(domain);
      if (cache) {
        cache.clear();
        this.persistenceManager.markDirty();
      }
    } else {
      // 使所有缓存无效
      this.clear();
    }
  }
  
  // 清空所有缓存
  clear(): void {
    this.caches.forEach(cache => cache.clear());
    this.persistenceManager.markDirty();
  }
  
  // 获取缓存统计
  getStats(): CacheStats {
    return this.statsCollector.getStats(this.caches);
  }
  
  // 清理缓存
  pruneCache(domain?: string): number {
    if (domain) {
      const cache = this.caches.get(domain);
      if (!cache) return 0;
      
      const strategy = this.strategyManager.getStrategy(domain);
      const keysToEvict = strategy.shouldEvict(cache);
      
      let count = 0;
      for (const key of keysToEvict) {
        if (cache.delete(key)) {
          count++;
        }
      }
      
      if (count > 0) {
        this.persistenceManager.markDirty();
      }
      
      return count;
    } else {
      // 清理所有域缓存
      let totalCount = 0;
      for (const domain of this.caches.keys()) {
        totalCount += this.pruneCache(domain);
      }
      return totalCount;
    }
  }
  
  // 持久化缓存
  async persist(): Promise<boolean> {
    return this.persistenceManager.persistCache();
  }
  
  // 加载缓存
  async load(): Promise<boolean> {
    try {
      const data = await this.persistenceManager.loadCache();
      
      // 分域加载缓存数据
      for (const [domain, entries] of Object.entries(data)) {
        const cache = this.getOrCreateDomainCache(domain);
        
        for (const [key, entry] of Object.entries(entries as Record<string, any>)) {
          cache.set(key, entry);
        }
      }
      
      return true;
    } catch (e) {
      console.error("加载缓存失败:", e);
      return false;
    }
  }
  
  // 获取或创建域缓存
  private getOrCreateDomainCache(domain: string): Map<string, CacheEntry<any>> {
    if (!this.caches.has(domain)) {
      this.caches.set(domain, new Map());
    }
    return this.caches.get(domain)!;
  }
  
  // 获取域的最大缓存大小
  private getMaxSizeForDomain(domain: string): number {
    switch (domain) {
      case 'displayNames':
        return this.settings.cacheSettings.maxDisplayNameEntries;
      case 'links':
        return this.settings.cacheSettings.maxLinkEntries;
      case 'decorations':
        return this.settings.cacheSettings.maxDecorationEntries;
      default:
        return 1000; // 默认大小
    }
  }
  
  // 初始化缓存域
  private initializeDomains(): void {
    // 预定义常用缓存域
    this.caches.set('displayNames', new Map());
    this.caches.set('links', new Map());
    this.caches.set('decorations', new Map());
  }
}
```
